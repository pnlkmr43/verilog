reset# Write 32-bit data to AXI slave
create_hw_axi_txn write_txn \
    [get_hw_axis] \
    -type WRITE \
    -address 40000000 \
    -data 12345678

# Execute the transaction
run_hw_axi [get_hw_axi_txns write_txn]



create_hw_axi_txn read_txn \
    [get_hw_axis] \
    -type READ \
    -address 41000000 \
    -len 1

run_hw_axi [get_hw_axi_txns read_txn]

# Display the read data
report_hw_axi_txn [get_hw_axi_txns read_txn]
